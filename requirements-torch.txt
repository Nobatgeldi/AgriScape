# older Python may use cu121; Python 3.13 needs a newer index such as cu128.

#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu128
#   pip install -r requirements-torch.txt
#
# For CPU-only inference, drop the --index-url line (plain `pip install torch
# torchvision`); deep_features.py also accepts --deep-device cpu.
